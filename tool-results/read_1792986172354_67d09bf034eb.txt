     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import {
     5→  Eye,
     6→  FolderHeart,
     7→  LayoutGrid,
     8→  Pencil,
     9→  RefreshCw,
    10→  SquarePen,
    11→  Trash2,
    12→} from "lucide-react";
    13→import { toast } from "sonner";
    14→
    15→import { api, formatDate } from "@/lib/client";
    16→import { navigate } from "@/lib/hash-router";
    17→import { useSession } from "@/lib/session";
    18→import type { ProjectCardDTO, PublicMe } from "@/lib/types";
    19→
    20→import { Button } from "@/components/ui/button";
    21→import { Input } from "@/components/ui/input";
    22→import { Label } from "@/components/ui/label";
    23→import { Textarea } from "@/components/ui/textarea";
    24→import { Badge } from "@/components/ui/badge";
    25→import { Skeleton } from "@/components/ui/skeleton";
    26→import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";
    27→import {
    28→  AlertDialog,
    29→  AlertDialogAction,
    30→  AlertDialogCancel,
    31→  AlertDialogContent,
    32→  AlertDialogDescription,
    33→  AlertDialogFooter,
    34→  AlertDialogHeader,
    35→  AlertDialogTitle,
    36→} from "@/components/ui/alert-dialog";
    37→import { EmptyState } from "@/components/bean/EmptyState";
    38→import { MediaThumb } from "@/components/bean/MediaThumb";
    39→import { ProfileHeader } from "@/components/bean/ProfileHeader";
    40→import { TagInput } from "@/components/bean/TagInput";
    41→import { StatCard, SectionHeader } from "@/components/bean/SectionHeader";
    42→import { categoryLabel } from "@/components/bean/ProjectCard";
    43→import type { MediaDTO, ProfessionDTO, SkillDTO } from "@/lib/types";
    44→
    45→// ------------------------------------------------------------------ types
    46→
    47→interface FolioData {
    48→  user: PublicMe;
    49→  stats: { projects: number; published: number; followers: number; following: number; profileViews: number };
    50→}
    51→
    52→interface CollectionItem {
    53→  id: string;
    54→  name: string;
    55→  description: string | null;
    56→  isPublic: boolean;
    57→  items: ProjectCardDTO[];
    58→}
    59→
    60→// ------------------------------------------------------------------ helpers
    61→
    62→function StatusBadge({ status }: { status: string }) {
    63→  return status === "PUBLISHED" ? (
    64→    <Badge className="rounded-full bg-gold-soft text-primary ring-1 ring-primary/25 border-transparent">Published</Badge>
    65→  ) : (
    66→    <Badge variant="secondary" className="rounded-full">Draft</Badge>
    67→  );
    68→}
    69→
    70→function VisibilityBadge({ visibility }: { visibility: string }) {
    71→  const label = visibility === "UNLISTED" ? "Unlisted" : visibility === "PRIVATE" ? "Private" : "Public";
    72→  return (
    73→    <Badge variant="outline" className="rounded-full text-muted-foreground">
    74→      {label}
    75→    </Badge>
    76→  );
    77→}
    78→
    79→function ProfileSkeleton() {
    80→  return (
    81→    <div className="space-y-6" aria-busy="true" aria-label="Loading your folio">
    82→      <div className="overflow-hidden rounded-xl border border-border/60 bg-card">
    83→        <Skeleton className="h-36 w-full rounded-none" />
    84→        <div className="flex items-end gap-4 p-6">
    85→          <Skeleton className="size-20 -mt-16 rounded-full" />
    86→          <div className="space-y-2 pb-1">
    87→            <Skeleton className="h-5 w-48" />
    88→            <Skeleton className="h-3 w-32" />
    89→          </div>
    90→        </div>
    91→      </div>
    92→      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-5">
    93→        {Array.from({ length: 5 }).map((_, i) => (
    94→          <Skeleton key={i} className="h-24 rounded-xl" />
    95→        ))}
    96→      </div>
    97→      <Skeleton className="h-72 rounded-xl" />
    98→    </div>
    99→  );
   100→}
   101→
   102→// ------------------------------------------------------------------ view
   103→
   104→export default function FolioView() {
   105→  const { refresh } = useSession();
   106→
   107→  const [loading, setLoading] = useState(true);
   108→  const [error, setError] = useState<string | null>(null);
   109→  const [data, setData] = useState<FolioData | null>(null);
   110→
   111→  // profile form
   112→  const [name, setName] = useState("");
   113→  const [headline, setHeadline] = useState("");
   114→  const [bio, setBio] = useState("");
   115→  const [location, setLocation] = useState("");
   116→  const [availability, setAvailability] = useState("");
   117→  const [contactEmail, setContactEmail] = useState("");
   118→  const [websiteUrl, setWebsiteUrl] = useState("");
   119→  const [professions, setProfessions] = useState<string[]>([]);
   120→  const [skills, setSkills] = useState<string[]>([]);
   121→  const [savingProfile, setSavingProfile] = useState(false);
   122→
   123→  const [allProfessions, setAllProfessions] = useState<ProfessionDTO[]>([]);
   124→  const [allSkills, setAllSkills] = useState<SkillDTO[]>([]);
   125→
   126→  // projects + collections
   127→  const [projects, setProjects] = useState<ProjectCardDTO[] | null>(null);
   128→  const [projectsLoading, setProjectsLoading] = useState(true);
   129→  const [collections, setCollections] = useState<CollectionItem[] | null>(null);
   130→  const [deleteTarget, setDeleteTarget] = useState<ProjectCardDTO | null>(null);
   131→  const [deleting, setDeleting] = useState(false);
   132→
   133→  // ------------------------------------------------------------ load
   134→
   135→  const load = useCallback(async () => {
   136→    setLoading(true);
   137→    setError(null);
   138→    try {
   139→      const res = await api<FolioData>("/api/me/profile");
   140→      setData(res);
   141→      setName(res.user.name);
   142→      setHeadline(res.user.headline ?? "");
   143→      setBio(res.user.bio ?? "");
   144→      setLocation(res.user.location ?? "");
   145→      setAvailability(res.user.availability ?? "");
   146→      setContactEmail(res.user.contactEmail ?? "");
   147→      setWebsiteUrl(res.user.websiteUrl ?? "");
   148→      setProfessions(res.user.professions ?? []);
   149→      setSkills(res.user.skills ?? []);
   150→    } catch (e) {
   151→      setError(e instanceof Error ? e.message : "Could not load your folio");
   152→    } finally {
   153→      setLoading(false);
   154→    }
   155→  }, []);
   156→
   157→  useEffect(() => {
   158→    void load();
   159→  }, [load]);
   160→
   161→  useEffect(() => {
   162→    api<{ professions: ProfessionDTO[] }>("/api/professions")
   163→      .then((r) => setAllProfessions(r.professions))
   164→      .catch(() => undefined);
   165→    api<{ skills: SkillDTO[] }>("/api/skills")
   166→      .then((r) => setAllSkills(r.skills))
   167→      .catch(() => undefined);
   168→  }, []);
   169→
   170→  const loadProjects = useCallback(async () => {
   171→    setProjectsLoading(true);
   172→    try {
   173→      const res = await api<{ projects: ProjectCardDTO[] }>("/api/projects");
   174→      setProjects(res.projects);
   175→    } catch {
   176→      // project list failures are non-fatal for the profile form
   177→      setProjects([]);
   178→    } finally {
   179→      setProjectsLoading(false);
   180→    }
   181→  }, []);
   182→
   183→  useEffect(() => {
   184→    void loadProjects();
   185→  }, [loadProjects]);
   186→
   187→  useEffect(() => {
   188→    api<{ collections: CollectionItem[] }>("/api/me/collections")
   189→      .then((r) => setCollections(r.collections))
   190→      .catch(() => setCollections([]));
   191→  }, []);
   192→
   193→  // ------------------------------------------------------------ actions
   194→
   195→  async function patchProfile(json: object, successMsg: string) {
   196→    const res = await api<{ user: PublicMe }>("/api/me/profile", { method: "PATCH", json });
   197→    setData((d) => (d ? { ...d, user: res.user } : d));
   198→    await refresh(); // keep the shell avatar/name in sync
   199→    toast.success(successMsg);
   200→    return res.user;
   201→  }
   202→
   203→  async function saveProfile() {
   204→    if (!name.trim()) {
   205→      toast.error("Your name can't be empty.");
   206→      return;
   207→    }
   208→    const url = websiteUrl.trim();
   209→    if (url && !/^https?:\/\/\S+$/i.test(url)) {
   210→      toast.error("Website must be a full http(s) URL, e.g. https://example.com");
   211→      return;
   212→    }
   213→    const email = contactEmail.trim();
   214→    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   215→      toast.error("Contact email doesn't look valid — leave it empty or use a full address.");
   216→      return;
   217→    }
   218→    setSavingProfile(true);
   219→    try {
   220→      await patchProfile(
   221→        {
   222→          name: name.trim(),
   223→          headline: headline.trim(),
   224→          bio: bio.trim(),
   225→          location: location.trim(),
   226→          availability: availability.trim(),
   227→          contactEmail: contactEmail.trim() || null,
   228→          websiteUrl: url || null,
   229→          professions,
   230→          skills,
   231→        },
   232→        "Profile saved"
   233→      );
   234→    } catch (e) {
   235→      toast.error(e instanceof Error ? e.message : "Could not save profile");
   236→    } finally {
   237→      setSavingProfile(false);
   238→    }
   239→  }
   240→
   241→  function toggleProfession(slug: string) {
   242→    setProfessions((cur) => {
   243→      if (cur.includes(slug)) return cur.filter((s) => s !== slug);
   244→      if (cur.length >= 5) {
   245→        toast.error("Pick up to 5 professions");
   246→        return cur;
   247→      }
   248→      return [...cur, slug];
   249→    });
   250→  }
   251→
   252→  async function onAvatarUploaded(m: MediaDTO) {
   253→    try {
   254→      await patchProfile({ avatarMediaId: m.id }, "Avatar updated");
   255→    } catch (e) {
   256→      toast.error(e instanceof Error ? e.message : "Could not set avatar");
   257→    }
   258→  }
   259→
   260→  async function onCoverUploaded(m: MediaDTO) {
   261→    try {
   262→      await patchProfile({ coverMediaId: m.id }, "Cover updated");
   263→    } catch (e) {
   264→      toast.error(e instanceof Error ? e.message : "Could not set cover");
   265→    }
   266→  }
   267→
   268→  async function confirmDelete() {
   269→    if (!deleteTarget) return;
   270→    setDeleting(true);
   271→    try {
   272→      await api(`/api/projects/${deleteTarget.id}`, { method: "DELETE" });
   273→      toast.success(`"${deleteTarget.title}" deleted`);
   274→      setDeleteTarget(null);
   275→      await loadProjects();
   276→    } catch (e) {
   277→      toast.error(e instanceof Error ? e.message : "Could not delete project");
   278→    } finally {
   279→      setDeleting(false);
   280→    }
   281→  }
   282→
   283→  // ------------------------------------------------------------ render guards
   284→
   285→  if (loading) {
   286→    return (
   287→      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6">
   288→        <ProfileSkeleton />
   289→      </div>
   290→    );
   291→  }
   292→
   293→  if (error || !data) {
   294→    return (
   295→      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6">
   296→        <EmptyState
   297→          icon={RefreshCw}
   298→          title="Could not load your folio"
   299→          description={error ?? "Something went wrong."}
   300→          action={<Button onClick={() => void load()}>Try again</Button>}
   301→        />
   302→      </div>
   303→    );
   304→  }
   305→
   306→  const u = data.user;
   307→
   308→  return (
   309→    <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6">
   310→      {/* ---------------- unified profile header (owner variant) */}
   311→      <ProfileHeader
   312→        variant="owner"
   313→        profile={{
   314→          name: u.name,
   315→          username: u.username,
   316→          headline: u.headline,
   317→          location: u.location,
   318→          availability: u.availability,
   319→          professions: u.professions,
   320→          skills: u.skills.map((slug) => allSkills.find((s) => s.slug === slug)?.label ?? slug),
   321→          avatarUrl: u.avatarUrl,
   322→          coverUrl: u.coverUrl,
   323→          coverPositionY: u.coverPositionY,
   324→          avatarFit: u.avatarFit,
   325→          avatarZoom: u.avatarZoom,
   326→          avatarPosX: u.avatarPosX,
   327→          avatarPosY: u.avatarPosY,
   328→        }}
   329→        ownerControls={{
   330→          onCoverUploaded: (m) => void onCoverUploaded(m),
   331→          onAvatarUploaded: (m) => void onAvatarUploaded(m),
   332→          onPatch: async (patch) => {
   333→            await patchProfile(patch, "Saved");
   334→          },
   335→          onRemoveCover: () => {
   336→            void patchProfile({ coverMediaId: null }, "Cover removed");
   337→          },
   338→          onRemoveAvatar: () => {
   339→            void patchProfile({ avatarMediaId: null }, "Avatar removed");
   340→          },
   341→        }}
   342→      />
   343→
   344→      {/* ---------------- stats */}
   345→      <section className="mt-6 grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-5" aria-label="Profile statistics">
   346→        <StatCard label="Projects" value={data.stats.projects} />
   347→        <StatCard label="Published" value={data.stats.published} />
   348→        <StatCard label="Followers" value={data.stats.followers} />
   349→        <StatCard label="Following" value={data.stats.following} />
   350→        <StatCard label="Profile views" value={data.stats.profileViews} hint="estimated" />
   351→      </section>
   352→
   353→      {/* ---------------- edit form */}
   354→      <section className="mt-6 rounded-xl border border-border/60 bg-card p-4 sm:p-6" aria-label="Edit profile">
   355→        <Tabs defaultValue="profile">
   356→          <TabsList>
   357→            <TabsTrigger value="profile">Profile</TabsTrigger>
   358→            <TabsTrigger value="work">Work</TabsTrigger>
   359→          </TabsList>
   360→
   361→          <TabsContent value="profile" className="mt-5">
   362→            <div className="grid gap-5 sm:grid-cols-2">
   363→              <div className="space-y-1.5">
   364→                <Label htmlFor="fp-name">Name</Label>
   365→                <Input id="fp-name" value={name} maxLength={80} onChange={(e) => setName(e.target.value)} />
   366→              </div>
   367→              <div className="space-y-1.5">
   368→                <Label htmlFor="fp-headline">Headline</Label>
   369→                <Input
   370→                  id="fp-headline"
   371→                  value={headline}
   372→                  maxLength={120}
   373→                  placeholder="e.g. Architect & set designer"
   374→                  onChange={(e) => setHeadline(e.target.value)}
   375→                />
   376→              </div>
   377→              <div className="space-y-1.5">
   378→                <Label htmlFor="fp-location">Location</Label>
   379→                <Input
   380→                  id="fp-location"
   381→                  value={location}
   382→                  maxLength={100}
   383→                  placeholder="e.g. Lisbon, Portugal"
   384→                  onChange={(e) => setLocation(e.target.value)}
   385→                />
   386→              </div>
   387→              <div className="space-y-1.5">
   388→                <Label htmlFor="fp-availability">Availability</Label>
   389→                <Input
   390→                  id="fp-availability"
   391→                  value={availability}
   392→                  maxLength={80}
   393→                  placeholder="e.g. Available for commissions"
   394→                  onChange={(e) => setAvailability(e.target.value)}
   395→                />
   396→              </div>
   397→              <div className="space-y-1.5">
   398→                <Label htmlFor="fp-email">Contact email</Label>
   399→                <Input
   400→                  id="fp-email"
   401→                  type="email"
   402→                  value={contactEmail}
   403→                  placeholder="Shown on your public profile for inquiries"
   404→                  onChange={(e) => setContactEmail(e.target.value)}
   405→                />
   406→              </div>
   407→              <div className="space-y-1.5">
   408→                <Label htmlFor="fp-website">Website</Label>
   409→                <Input
   410→                  id="fp-website"
   411→                  type="url"
   412→                  value={websiteUrl}
   413→                  placeholder="https://your-site.com"
   414→                  onChange={(e) => setWebsiteUrl(e.target.value)}
   415→                />
   416→              </div>
   417→              <div className="space-y-1.5 sm:col-span-2">
   418→                <Label htmlFor="fp-bio">Bio</Label>
   419→                <Textarea
   420→                  id="fp-bio"
   421→                  value={bio}
   422→                  maxLength={2000}
   423→                  placeholder="A short introduction — what you make, who you work with, what you're looking for."
   424→                  className="min-h-28"
   425→                  onChange={(e) => setBio(e.target.value)}
   426→                />
   427→                <p className="text-right text-xs text-muted-foreground">{bio.length}/2000</p>
   428→              </div>
   429→            </div>
   430→          </TabsContent>
   431→
   432→          <TabsContent value="work" className="mt-5">
   433→            <div className="space-y-6">
   434→              <div className="space-y-1.5">
   435→                <Label>Professions</Label>
   436→                <p className="text-xs text-muted-foreground">Pick up to 5 — these power Discover recommendations.</p>
   437→                <div className="flex flex-wrap gap-2">
   438→                  {allProfessions.length === 0 && (
   439→                    <div className="flex gap-2">
   440→                      <Skeleton className="h-7 w-20 rounded-full" />
   441→                      <Skeleton className="h-7 w-24 rounded-full" />
   442→                      <Skeleton className="h-7 w-16 rounded-full" />
   443→                    </div>
   444→                  )}
   445→                  {allProfessions.map((p) => {
   446→                    const active = professions.includes(p.slug);
   447→                    return (
   448→                      <button
   449→                        key={p.slug}
   450→                        type="button"
   451→                        aria-pressed={active}
   452→                        onClick={() => toggleProfession(p.slug)}
   453→                        className={
   454→                          active
   455→                            ? "rounded-full bg-primary px-3 py-1.5 text-xs font-medium text-primary-foreground transition-opacity hover:opacity-90"
   456→                            : "rounded-full border border-border bg-background px-3 py-1.5 text-xs text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground"
   457→                        }
   458→                      >
   459→                        {p.label}
   460→                      </button>
   461→                    );
   462→                  })}
   463→                </div>
   464→              </div>
   465→              <div className="space-y-1.5">
   466→                <Label htmlFor="fp-skills">Skills</Label>
   467→                <TagInput
   468→                  value={skills}
   469→                  onChange={setSkills}
   470→                  max={20}
   471→                  placeholder="e.g. Rendering, Copywriting, Color grading…"
   472→                  ariaLabel="Add skills"
   473→                />
   474→                <p className="text-xs text-muted-foreground">Press Enter or comma to add · max 20</p>
   475→              </div>
   476→            </div>
   477→          </TabsContent>
   478→        </Tabs>
   479→
   480→        <div className="mt-6 flex justify-end border-t border-border/50 pt-4">
   481→          <Button onClick={() => void saveProfile()} disabled={savingProfile}>
   482→            {savingProfile ? "Saving…" : "Save profile"}
   483→          </Button>
   484→        </div>
   485→      </section>
   486→
   487→      {/* ---------------- projects management */}
   488→      <section className="mt-10" aria-label="Your projects">
   489→        <SectionHeader
   490→          title="Your projects"
   491→          description="Everything you've created — drafts and published."
   492→          action={
   493→            <Button variant="outline" size="sm" onClick={() => navigate("/studio")}>
   494→              <LayoutGrid className="size-4" aria-hidden /> Manage in Studio
   495→            </Button>
   496→          }
   497→        />
   498→
   499→        <div className="mt-5">
   500→          {projectsLoading && (
   501→            <div className="space-y-3" aria-busy="true">
   502→              {Array.from({ length: 2 }).map((_, i) => (
   503→                <div key={i} className="flex items-center gap-4 rounded-xl border border-border/60 bg-card p-4">
   504→                  <Skeleton className="h-[63px] w-28 rounded-lg" />
   505→                  <div className="flex-1 space-y-2">
   506→                    <Skeleton className="h-4 w-2/5" />
   507→                    <Skeleton className="h-3 w-1/4" />
   508→                  </div>
   509→                  <Skeleton className="hidden h-8 w-24 sm:block" />
   510→                </div>
   511→              ))}
   512→            </div>
   513→          )}
   514→
   515→          {!projectsLoading && projects !== null && projects.length === 0 && (
   516→            <EmptyState
   517→              icon={SquarePen}
   518→              title="No projects yet"
   519→              description="Start your first project in the studio — from brief to published piece."
   520→              action={<Button onClick={() => navigate("/studio/new")}>New project</Button>}
   521→            />
   522→          )}
   523→
   524→          {!projectsLoading && projects !== null && projects.length > 0 && (
   525→            <div className="space-y-3">
   526→              {projects.map((p) => (
   527→                <div
   528→                  key={p.id}
   529→                  className="flex flex-col gap-3 rounded-xl border border-border/60 bg-card p-4 transition-colors hover:border-primary/25 sm:flex-row sm:items-center"
   530→                >
   531→                  <button
   532→                    type="button"
   533→                    className="w-full shrink-0 sm:w-28"
   534→                    onClick={() => navigate(`/p/${p.id}`)}
   535→                    aria-label={`Open ${p.title}`}
   536→                  >
   537→                    <MediaThumb src={p.coverUrl} alt={p.title} aspect="aspect-[16/9]" className="rounded-lg sm:w-28" />
   538→                  </button>
   539→                  <div className="min-w-0 flex-1">
   540→                    <button
   541→                      type="button"
   542→                      onClick={() => navigate(`/p/${p.id}`)}
   543→                      className="line-clamp-1 text-left font-medium transition-colors hover:text-primary"
   544→                    >
   545→                      {p.title}
   546→                    </button>
   547→                    <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   548→                      <StatusBadge status={p.status} />
   549→                      <VisibilityBadge visibility={p.visibility} />
   550→                      <span className="text-xs text-muted-foreground">{categoryLabel(p.category)}</span>
   551→                    </div>
   552→                    <p className="mt-1 flex flex-wrap items-center gap-x-3 text-xs text-muted-foreground">
   553→                      <span>Updated {formatDate(p.updatedAt)}</span>
   554→                      <span className="inline-flex items-center gap-1">
   555→                        <Eye className="size-3" aria-hidden /> {p.counts.views} views (est.)
   556→                      </span>
   557→                    </p>
   558→                  </div>
   559→                  <div className="flex flex-wrap items-center gap-1.5 sm:shrink-0 sm:justify-end">
   560→                    <Button variant="outline" size="sm" onClick={() => navigate(`/studio/${p.id}/edit`)}>
   561→                      <Pencil className="size-3.5" aria-hidden /> Edit
   562→                    </Button>
   563→                    <Button variant="ghost" size="sm" onClick={() => navigate(`/p/${p.id}`)}>
   564→                      View
   565→                    </Button>
   566→                    <Button
   567→                      variant="ghost"
   568→                      size="icon"
   569→                      className="size-8 text-muted-foreground hover:text-destructive"
   570→                      aria-label={`Delete ${p.title}`}
   571→                      onClick={() => setDeleteTarget(p)}
   572→                    >
   573→                      <Trash2 className="size-3.5" aria-hidden />
   574→                    </Button>
   575→                  </div>
   576→                </div>
   577→              ))}
   578→            </div>
   579→          )}
   580→        </div>
   581→      </section>
   582→
   583→      {/* ---------------- collections */}
   584→      <section className="mt-10" aria-label="Your collections">
   585→        <SectionHeader
   586→          title="Public collections"
   587→          description="Curated sets of saved work on your profile."
   588→          action={
   589→            <Button variant="outline" size="sm" onClick={() => navigate("/collections")}>
   590→              <FolderHeart className="size-4" aria-hidden /> Manage collections
   591→            </Button>
   592→          }
   593→        />
   594→        <div className="mt-5">
   595→          {collections === null && (
   596→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3" aria-busy="true">
   597→              {Array.from({ length: 3 }).map((_, i) => (
   598→                <Skeleton key={i} className="h-28 rounded-xl" />

   600→            </div>
   601→          )}
   602→          {collections !== null && collections.length === 0 && (
   603→            <EmptyState
   604→              icon={FolderHeart}
   605→              title="No collections yet"
   606→              description="Group the work you save into named collections and share them from your profile."
   607→              action={<Button variant="outline" onClick={() => navigate("/collections")}>Go to Saves & collections</Button>}
   608→            />
   609→          )}
   610→          {collections !== null && collections.length > 0 && (
   611→            <ul className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   612→              {collections.map((c) => (
   613→                <li key={c.id}>
   614→                  <button
   615→                    type="button"
   616→                    onClick={() => navigate("/collections")}
   617→                    className="h-full w-full rounded-xl border border-border/60 bg-card p-4 text-left transition-colors hover:border-primary/25"
   618→                  >
   619→                    <div className="flex items-center justify-between gap-2">

   621→                      <Badge variant="secondary" className="rounded-full">
   622→                        {c.items.length} {c.items.length === 1 ? "item" : "items"}
   623→                      </Badge>
   624→                    </div>

   627→                  </button>
   628→                </li>
   629→              ))}
   630→            </ul>
   631→          )}
   632→        </div>
   633→      </section>
   634→
   635→      {/* ---------------- delete confirm */}
   636→      <AlertDialog open={deleteTarget !== null} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   637→        <AlertDialogContent>
   638→          <AlertDialogHeader>
   639→            <AlertDialogTitle>Delete “{deleteTarget?.title}”?</AlertDialogTitle>
   640→            <AlertDialogDescription>
   641→              This permanently removes the project and its media links. This action cannot be undone.
   642→            </AlertDialogDescription>
   643→          </AlertDialogHeader>
   644→          <AlertDialogFooter>
   645→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   646→            <AlertDialogAction
   647→              className="bg-destructive text-white hover:bg-destructive/90"
   648→              disabled={deleting}
   649→              onClick={(e) => {
   650→                e.preventDefault();
   651→                void confirmDelete();
   652→              }}
   653→            >
   654→              {deleting ? "Deleting…" : "Delete project"}
   655→            </AlertDialogAction>
   656→          </AlertDialogFooter>
   657→        </AlertDialogContent>
   658→      </AlertDialog>
   659→    </div>
   660→  );
