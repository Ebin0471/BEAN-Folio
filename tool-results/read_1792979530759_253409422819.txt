    69→function StatusBadge({ status }: { status: string }) {
    70→  return status === "PUBLISHED" ? (
    71→    <Badge className="rounded-full bg-gold-soft text-primary ring-1 ring-primary/25 border-transparent">Published</Badge>
    72→  ) : (
    73→    <Badge variant="secondary" className="rounded-full">Draft</Badge>
    74→  );
    75→}
    76→
    77→function VisibilityBadge({ visibility }: { visibility: string }) {
    78→  const label = visibility === "UNLISTED" ? "Unlisted" : visibility === "PRIVATE" ? "Private" : "Public";
    79→  return (
    80→    <Badge variant="outline" className="rounded-full text-muted-foreground">
    81→      {label}
    82→    </Badge>
    83→  );
    84→}

    95→            <Skeleton className="h-3 w-32" />
    96→          </div>
    97→        </div>
    98→      </div>
    99→      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-5">
   100→        {Array.from({ length: 5 }).map((_, i) => (
   101→          <Skeleton key={i} className="h-24 rounded-xl" />

   106→  );

   109→// ------------------------------------------------------------------ view
   110→
   111→export default function FolioView() {
   112→  const { refresh } = useSession();
   113→
   114→  const [loading, setLoading] = useState(true);
   115→  const [error, setError] = useState<string | null>(null);
   116→  const [data, setData] = useState<FolioData | null>(null);
   117→
   118→  // profile form
   119→  const [name, setName] = useState("");
   120→  const [headline, setHeadline] = useState("");
   121→  const [bio, setBio] = useState("");
   122→  const [location, setLocation] = useState("");
   123→  const [availability, setAvailability] = useState("");
   124→  const [contactEmail, setContactEmail] = useState("");
   125→  const [websiteUrl, setWebsiteUrl] = useState("");
   126→  const [professions, setProfessions] = useState<string[]>([]);
   127→  const [skills, setSkills] = useState<string[]>([]);
   128→  const [savingProfile, setSavingProfile] = useState(false);
   129→
   130→  const [allProfessions, setAllProfessions] = useState<ProfessionDTO[]>([]);
   131→
   132→  // projects + collections
   133→  const [projects, setProjects] = useState<ProjectCardDTO[] | null>(null);
   134→  const [projectsLoading, setProjectsLoading] = useState(true);
   135→  const [collections, setCollections] = useState<CollectionItem[] | null>(null);
   136→  const [deleteTarget, setDeleteTarget] = useState<ProjectCardDTO | null>(null);
   137→  const [deleting, setDeleting] = useState(false);
   138→
   139→  // ------------------------------------------------------------ load
   140→
   141→  const load = useCallback(async () => {
   142→    setLoading(true);
   143→    setError(null);
   144→    try {
   145→      const res = await api<FolioData>("/api/me/profile");
   146→      setData(res);
   147→      setName(res.user.name);
   148→      setHeadline(res.user.headline ?? "");
   149→      setBio(res.user.bio ?? "");
   150→      setLocation(res.user.location ?? "");
   151→      setAvailability(res.user.availability ?? "");
   152→      setContactEmail(res.user.contactEmail ?? "");
   153→      setWebsiteUrl(res.user.websiteUrl ?? "");
   154→      setProfessions(res.user.professions ?? []);
   155→      setSkills(res.user.skills ?? []);
   156→    } catch (e) {
   157→      setError(e instanceof Error ? e.message : "Could not load your folio");
   158→    } finally {
   159→      setLoading(false);
   160→    }
   161→  }, []);
   162→
   163→  useEffect(() => {
   164→    void load();
   165→  }, [load]);
   166→
   167→  useEffect(() => {
   168→    api<{ professions: ProfessionDTO[] }>("/api/professions")
   169→      .then((r) => setAllProfessions(r.professions))
   170→      .catch(() => undefined);
   171→  }, []);
   172→
   173→  const loadProjects = useCallback(async () => {
   174→    setProjectsLoading(true);
   175→    try {
   176→      const res = await api<{ projects: ProjectCardDTO[] }>("/api/projects");
   177→      setProjects(res.projects);
   178→    } catch {
   179→      // project list failures are non-fatal for the profile form
   180→      setProjects([]);
   181→    } finally {
   182→      setProjectsLoading(false);
   183→    }
   184→  }, []);
   185→
   186→  useEffect(() => {
   187→    void loadProjects();
   188→  }, [loadProjects]);
   189→
   190→  useEffect(() => {
   191→    api<{ collections: CollectionItem[] }>("/api/me/collections")
   192→      .then((r) => setCollections(r.collections))
   193→      .catch(() => setCollections([]));
   194→  }, []);
   195→
   196→  // ------------------------------------------------------------ actions
   197→
   198→  async function patchProfile(json: Record<string, unknown>, successMsg: string) {
   199→    const res = await api<{ user: PublicMe }>("/api/me/profile", { method: "PATCH", json });
   200→    setData((d) => (d ? { ...d, user: res.user } : d));
   201→    await refresh(); // keep the shell avatar/name in sync
   202→    toast.success(successMsg);
   203→    return res.user;
   204→  }
   205→
   206→  async function saveProfile() {
   207→    if (!name.trim()) {
   208→      toast.error("Your name can't be empty.");
   209→      return;
   210→    }
   211→    const url = websiteUrl.trim();
   212→    if (url && !/^https?:\/\/\S+$/i.test(url)) {
   213→      toast.error("Website must be a full http(s) URL, e.g. https://example.com");
   214→      return;
   215→    }
   216→    const email = contactEmail.trim();
   217→    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   218→      toast.error("Contact email doesn't look valid — leave it empty or use a full address.");
   219→      return;
   220→    }
   221→    setSavingProfile(true);
   222→    try {
   223→      await patchProfile(
   224→        {
   225→          name: name.trim(),
   226→          headline: headline.trim(),
   227→          bio: bio.trim(),
   228→          location: location.trim(),
   229→          availability: availability.trim(),
   230→          contactEmail: contactEmail.trim() || null,
   231→          websiteUrl: url || null,
   232→          professions,
   233→          skills,
   234→        },
   235→        "Profile saved"
   236→      );
   237→    } catch (e) {
   238→      toast.error(e instanceof Error ? e.message : "Could not save profile");
   239→    } finally {
   240→      setSavingProfile(false);
   241→    }
   242→  }
   243→
   244→  function toggleProfession(slug: string) {
   245→    setProfessions((cur) => {
   246→      if (cur.includes(slug)) return cur.filter((s) => s !== slug);
   247→      if (cur.length >= 5) {
   248→        toast.error("Pick up to 5 professions");
   249→        return cur;
   250→      }
   251→      return [...cur, slug];
   252→    });
   253→  }
   254→
   255→  async function onAvatarUploaded(m: MediaDTO) {
   256→    try {
   257→      await patchProfile({ avatarMediaId: m.id }, "Avatar updated");
   258→    } catch (e) {
   259→      toast.error(e instanceof Error ? e.message : "Could not set avatar");
   260→    }
   261→  }
   262→
   263→  async function onCoverUploaded(m: MediaDTO) {
   264→    try {
   265→      await patchProfile({ coverMediaId: m.id }, "Cover updated");
   266→    } catch (e) {
   267→      toast.error(e instanceof Error ? e.message : "Could not set cover");
   268→    }
   269→  }
   270→
   271→  async function confirmDelete() {
   272→    if (!deleteTarget) return;
   273→    setDeleting(true);
   274→    try {
   275→      await api(`/api/projects/${deleteTarget.id}`, { method: "DELETE" });
   276→      toast.success(`"${deleteTarget.title}" deleted`);
   277→      setDeleteTarget(null);
   278→      await loadProjects();
   279→    } catch (e) {
   280→      toast.error(e instanceof Error ? e.message : "Could not delete project");
   281→    } finally {
   282→      setDeleting(false);
   283→    }
   284→  }
   285→
   286→  // ------------------------------------------------------------ render guards
   287→
   288→  if (loading) {
   289→    return (
   290→      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6">
   291→        <ProfileSkeleton />
   292→      </div>
   293→    );
   294→  }
   295→
   296→  if (error || !data) {
   297→    return (
   298→      <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6">
   299→        <EmptyState
   300→          icon={RefreshCw}
   301→          title="Could not load your folio"
   302→          description={error ?? "Something went wrong."}
   303→          action={<Button onClick={() => void load()}>Try again</Button>}
   304→        />
   305→      </div>
   306→    );
   307→  }
   308→
   309→  const u = data.user;
   310→  const initials = u.name
   311→    .split(" ")
   312→    .map((w) => w[0])
   313→    .slice(0, 2)
   314→    .join("")
   315→    .toUpperCase();
   316→
   317→  return (
   318→    <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6">
   319→      {/* ---------------- profile header card */}
   320→      <section className="overflow-hidden rounded-xl border border-border/60 bg-card" aria-label="Profile header">
   321→        <div className="relative h-36 bg-muted sm:h-40">
   322→          {u.coverUrl ? (
   323→            <img src={u.coverUrl} alt={`${u.name}'s cover`} className="h-full w-full object-cover" />
   324→          ) : (
   325→            <div className="grid h-full w-full place-items-center bg-gradient-to-br from-muted to-card text-muted-foreground">
   326→              <span className="text-xs">No cover image yet</span>
   327→            </div>
   328→          )}
   329→          <div className="absolute right-3 top-3">
   330→            <MediaUploadButton
   331→              kind="IMAGE"
   332→              label="Change cover"
   333→              onUploaded={(m) => void onCoverUploaded(m)}
   334→              variant="outline"
   335→              size="sm"
   336→              className="h-8 bg-background/70 text-xs backdrop-blur-sm"
   337→            />
   338→          </div>
   339→        </div>
   340→
   341→        <div className="px-4 pb-5 sm:px-6">
   342→          <div className="flex flex-wrap items-start justify-between gap-4">
   343→            <div className="flex items-start gap-4">
   344→              <div className="-mt-10 flex flex-col items-center gap-1">
   345→                <span className="block size-20 overflow-hidden rounded-full ring-2 ring-background">
   346→                  {u.avatarUrl ? (
   347→                    <img src={u.avatarUrl} alt={`${u.name}'s avatar`} className="h-full w-full object-cover" />
   348→                  ) : (
   349→                    <span className="grid h-full w-full place-items-center rounded-full bg-gold-soft font-display text-xl text-primary">
   350→                      {initials}
   351→                    </span>
   352→                  )}
   353→                </span>
   354→                <MediaUploadButton
   355→                  kind="IMAGE"
   356→                  label="Change avatar"
   357→                  onUploaded={(m) => void onAvatarUploaded(m)}
   358→                  variant="ghost"
   359→                  size="sm"
   360→                  className="h-7 px-2 text-xs text-muted-foreground"
   361→                />
   362→              </div>
   363→              <div className="pb-1">
   364→                <h1 className="font-display text-2xl font-semibold tracking-tight">{u.name}</h1>
   365→                <p className="text-sm text-muted-foreground">
   366→                  {u.username ? `@${u.username}` : u.email}
   367→                  {u.availability && (
   368→                    <span className="ml-2 inline-block rounded-full bg-gold-soft px-2 py-0.5 text-xs font-medium text-primary ring-1 ring-primary/25">
   369→                      {u.availability}
   370→                    </span>
   371→                  )}
   372→                </p>
   373→              </div>
   374→            </div>
   375→
   376→            <div className="flex flex-wrap items-center gap-2 pb-1">
   377→              <TooltipProvider>
   378→                <Tooltip>
   379→                  <TooltipTrigger asChild>
   380→                    <span tabIndex={u.username ? 0 : -1}>
   381→                      <Button
   382→                        variant="outline"
   383→                        disabled={!u.username}
   384→                        onClick={() => u.username && navigate(`/u/${u.username}`)}
   385→                      >
   386→                        <UserRound className="size-4" aria-hidden /> View public profile
   387→                      </Button>
   388→                    </span>
   389→                  </TooltipTrigger>
   390→                  {!u.username && <TooltipContent>Finish onboarding to claim your username</TooltipContent>}
   391→                </Tooltip>
   392→              </TooltipProvider>
   393→              <Button variant="ghost" onClick={() => navigate("/studio")}>
   394→                <SquarePen className="size-4" aria-hidden /> Studio
   395→              </Button>
   396→            </div>
   397→          </div>
   398→        </div>
   399→      </section>
   400→
   401→      {/* ---------------- stats */}
   402→      <section className="mt-6 grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-5" aria-label="Profile statistics">
   403→        <StatCard label="Projects" value={data.stats.projects} />
   404→        <StatCard label="Published" value={data.stats.published} />
   405→        <StatCard label="Followers" value={data.stats.followers} />
   406→        <StatCard label="Following" value={data.stats.following} />
   407→        <StatCard label="Profile views" value={data.stats.profileViews} hint="estimated" />
   408→      </section>
   409→
   410→      {/* ---------------- edit form */}
   411→      <section className="mt-6 rounded-xl border border-border/60 bg-card p-4 sm:p-6" aria-label="Edit profile">
   412→        <Tabs defaultValue="profile">
   413→          <TabsList>
   414→            <TabsTrigger value="profile">Profile</TabsTrigger>
   415→            <TabsTrigger value="work">Work</TabsTrigger>
   416→          </TabsList>
   417→
   418→          <TabsContent value="profile" className="mt-5">
   419→            <div className="grid gap-5 sm:grid-cols-2">
   420→              <div className="space-y-1.5">
   421→                <Label htmlFor="fp-name">Name</Label>
   422→                <Input id="fp-name" value={name} maxLength={80} onChange={(e) => setName(e.target.value)} />
   423→              </div>
   424→              <div className="space-y-1.5">
   425→                <Label htmlFor="fp-headline">Headline</Label>
   426→                <Input
   427→                  id="fp-headline"
   428→                  value={headline}
   429→                  maxLength={120}
   430→                  placeholder="e.g. Architect & set designer"
   431→                  onChange={(e) => setHeadline(e.target.value)}
   432→                />
   433→              </div>
   434→              <div className="space-y-1.5">
   435→                <Label htmlFor="fp-location">Location</Label>
   436→                <Input
   437→                  id="fp-location"
   438→                  value={location}
   439→                  maxLength={100}
   440→                  placeholder="e.g. Lisbon, Portugal"
   441→                  onChange={(e) => setLocation(e.target.value)}
   442→                />
   443→              </div>
   444→              <div className="space-y-1.5">
   445→                <Label htmlFor="fp-availability">Availability</Label>
   446→                <Input
   447→                  id="fp-availability"
   448→                  value={availability}
   449→                  maxLength={80}
   450→                  placeholder="e.g. Available for commissions"
   451→                  onChange={(e) => setAvailability(e.target.value)}
   452→                />
   453→              </div>
   454→              <div className="space-y-1.5">
   455→                <Label htmlFor="fp-email">Contact email</Label>
   456→                <Input
   457→                  id="fp-email"
   458→                  type="email"
   459→                  value={contactEmail}
   460→                  placeholder="Shown on your public profile for inquiries"
   461→                  onChange={(e) => setContactEmail(e.target.value)}
   462→                />
   463→              </div>
   464→              <div className="space-y-1.5">
   465→                <Label htmlFor="fp-website">Website</Label>
   466→                <Input
   467→                  id="fp-website"
   468→                  type="url"
   469→                  value={websiteUrl}
   470→                  placeholder="https://your-site.com"
   471→                  onChange={(e) => setWebsiteUrl(e.target.value)}
   472→                />
   473→              </div>
   474→              <div className="space-y-1.5 sm:col-span-2">
   475→                <Label htmlFor="fp-bio">Bio</Label>
   476→                <Textarea
   477→                  id="fp-bio"
   478→                  value={bio}
   479→                  maxLength={2000}
   480→                  placeholder="A short introduction — what you make, who you work with, what you're looking for."
   481→                  className="min-h-28"
   482→                  onChange={(e) => setBio(e.target.value)}
   483→                />
   484→                <p className="text-right text-xs text-muted-foreground">{bio.length}/2000</p>
   485→              </div>
   486→            </div>
   487→          </TabsContent>
   488→
   489→          <TabsContent value="work" className="mt-5">
   490→            <div className="space-y-6">
   491→              <div className="space-y-1.5">
   492→                <Label>Professions</Label>
   493→                <p className="text-xs text-muted-foreground">Pick up to 5 — these power Discover recommendations.</p>
   494→                <div className="flex flex-wrap gap-2">
   495→                  {allProfessions.length === 0 && (
   496→                    <div className="flex gap-2">
   497→                      <Skeleton className="h-7 w-20 rounded-full" />
   498→                      <Skeleton className="h-7 w-24 rounded-full" />
   499→                      <Skeleton className="h-7 w-16 rounded-full" />
   500→                    </div>
   501→                  )}
   502→                  {allProfessions.map((p) => {
   503→                    const active = professions.includes(p.slug);
   504→                    return (
   505→                      <button
   506→                        key={p.slug}
   507→                        type="button"
   508→                        aria-pressed={active}
   509→                        onClick={() => toggleProfession(p.slug)}
   510→                        className={
   511→                          active
   512→                            ? "rounded-full bg-primary px-3 py-1.5 text-xs font-medium text-primary-foreground transition-opacity hover:opacity-90"
   513→                            : "rounded-full border border-border bg-background px-3 py-1.5 text-xs text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground"
   514→                        }
   515→                      >
   516→                        {p.label}
   517→                      </button>
   518→                    );
   519→                  })}
   520→                </div>
   521→              </div>
   522→              <div className="space-y-1.5">
   523→                <Label htmlFor="fp-skills">Skills</Label>
   524→                <TagInput
   525→                  value={skills}
   526→                  onChange={setSkills}
   527→                  max={20}
   528→                  placeholder="e.g. Rendering, Copywriting, Color grading…"
   529→                  ariaLabel="Add skills"
   530→                />
   531→                <p className="text-xs text-muted-foreground">Press Enter or comma to add · max 20</p>
   532→              </div>
   533→            </div>
   534→          </TabsContent>
   535→        </Tabs>
   536→
   537→        <div className="mt-6 flex justify-end border-t border-border/50 pt-4">
   538→          <Button onClick={() => void saveProfile()} disabled={savingProfile}>
   539→            {savingProfile ? "Saving…" : "Save profile"}
   540→          </Button>
   541→        </div>
   542→      </section>
   543→
   544→      {/* ---------------- projects management */}
   545→      <section className="mt-10" aria-label="Your projects">
   546→        <SectionHeader
   547→          title="Your projects"
   548→          description="Everything you've created — drafts and published."
   549→          action={
   550→            <Button variant="outline" size="sm" onClick={() => navigate("/studio")}>
   551→              <LayoutGrid className="size-4" aria-hidden /> Manage in Studio
   552→            </Button>
   553→          }
   554→        />
   555→
   556→        <div className="mt-5">
   557→          {projectsLoading && (
   558→            <div className="space-y-3" aria-busy="true">
   559→              {Array.from({ length: 2 }).map((_, i) => (
   560→                <div key={i} className="flex items-center gap-4 rounded-xl border border-border/60 bg-card p-4">
   561→                  <Skeleton className="h-[63px] w-28 rounded-lg" />
   562→                  <div className="flex-1 space-y-2">
   563→                    <Skeleton className="h-4 w-2/5" />
   564→                    <Skeleton className="h-3 w-1/4" />
   565→                  </div>
