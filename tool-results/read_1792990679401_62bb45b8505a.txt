    57→// ------------------------------------------------------------------ preview frame
    58→// Renders the real SiteRenderer inside an iframe so Tailwind breakpoints respond
    59→// to the DEVICE width, not the editor viewport — an authentic mobile/tablet/desktop
    60→// preview rather than a squeezed div.
    61→
    62→function PreviewFrame({
    63→  width,
    64→  height,
    65→  label,
    66→  children,
    67→}: {
    68→  width: number | "100%";
    69→  height: string;
    70→  label: string;
    71→  children: React.ReactNode;
    72→}) {
    73→  const frameRef = useRef<HTMLIFrameElement>(null);
    74→  const rootRef = useRef<ReturnType<typeof import("react-dom/client").createRoot> | null>(null);
    75→  const [doc, setDoc] = useState<Document | null>(null);
    76→
    77→  const handleLoad = useCallback(() => {
    78→    const frame = frameRef.current;
    79→    const frameDoc = frame?.contentDocument;
    80→    if (!frameDoc) return;
    81→    // Copy the app's stylesheets into the frame so Tailwind classes work inside.
    82→    if (!frameDoc.head.hasChildNodes()) {
    83→      for (const node of Array.from(document.querySelectorAll('link[rel="stylesheet"], style'))) {
    84→        frameDoc.head.appendChild(node.cloneNode(true));
    85→      }
    86→      frameDoc.body.style.margin = "0";
    87→      frameDoc.body.style.backgroundColor = "transparent";
    88→    }
    89→    setDoc(frameDoc);
    90→  }, []);
    91→
    92→  useEffect(() => {
    93→    if (!doc) return;
    94→    let cancelled = false;
    95→    (async () => {
    96→      const ReactDOMClient = await import("react-dom/client");
    97→      if (cancelled) return;
    98→      if (!rootRef.current) {
    99→        const mount = doc.createElement("div");
   100→        doc.body.appendChild(mount);
   101→        rootRef.current = ReactDOMClient.createRoot(mount);
   102→      }
   103→      rootRef.current.render(children);
   104→    })();
   105→    return () => {
   106→      cancelled = true;
   107→    };
   108→  }, [doc, children]);
   109→
   110→  useEffect(() => {
   111→    return () => {
   112→      const root = rootRef.current;
   113→      rootRef.current = null;
   114→      if (root) {
   115→        // root.unmount() is illegal inside React's render/commit window (it
   116→        // throws "Attempted to synchronously unmount a root while React was
   117→        // already rendering" when the editor page is being swapped out) —
   118→        // always defer it past the current task.
   119→        setTimeout(() => {
   120→          try {
   121→            root.unmount();
   122→          } catch {
   123→            /* already gone */
   124→          }
   125→        }, 0);
   126→      }
   127→    };
   128→  }, []);
   129→
   130→  return (
   131→    <iframe
   132→      ref={frameRef}
   133→      title={label}
   134→      onLoad={handleLoad}
   135→      srcDoc="<!DOCTYPE html><html><head></head><body></body></html>"
   136→      className="block border border-border/70 bg-background shadow-sm"
   137→      style={{ width, height, maxWidth: "100%", margin: "0 auto" }}
   138→      sandbox="allow-same-origin allow-scripts"
   139→    />
   140→  );
   141→}

   161→function SectionSummary({ section }: { section: SiteSection }) {
   162→  switch (section.type) {
   163→    case "hero":
   164→      return section.subtitle || "Opening headline";
   165→    case "about":
   166→      return section.body ? section.body.slice(0, 60) : "Story and bio";
   167→    case "projects":
   168→      return `Up to ${section.limit} projects · ${section.layout}`;
   169→    case "collections":
   170→      return `Up to ${section.limit} collections`;
   171→    case "contact":
   172→      return section.body ? section.body.slice(0, 60) : "Contact details";
   173→    default:
   174→      return "Profile data section";
   175→  }
   176→}

   246→  const mutateConfig = useCallback((fn: (c: SiteConfig) => SiteConfig) => {
   247→    setConfig((c) => (c ? fn(c) : c));
   248→    setDirty(true);

   251→  function addSection(type: SiteSectionType) {
   252→    if (!config) return;
   253→    if (config.sections.length >= 12) {
   254→      toast.error("A website can have at most 12 sections.");
   255→      return;
   256→    }
   257→    const section = makeSection(type);
   258→    mutateConfig((c) => ({ ...c, sections: [...c.sections, section] }));
   259→    setSelectedSectionId(section.id);
   260→    setPanelTab("section");
   261→  }
   262→
   263→  function duplicateSection(id: string) {
   264→    if (!config) return;
   265→    const idx = config.sections.findIndex((s) => s.id === id);
   266→    if (idx === -1) return;
   267→    if (config.sections.length >= 12) {
   268→      toast.error("A website can have at most 12 sections.");
   269→      return;
   270→    }
   271→    const copy: SiteSection = { ...config.sections[idx], id: `${config.sections[idx].type}-${Date.now().toString(36)}-copy` };
   272→    const sections = [...config.sections];
   273→    sections.splice(idx + 1, 0, copy);
   274→    mutateConfig((c) => ({ ...c, sections }));
   275→    setSelectedSectionId(copy.id);
   276→  }
   277→
   278→  function removeSection(id: string) {
   279→    if (!config) return;
   280→    if (config.sections.length <= 1) {
   281→      toast.error("A website needs at least one section.");
   282→      return;
   283→    }
   284→    mutateConfig((c) => ({ ...c, sections: c.sections.filter((s) => s.id !== id) }));
   285→    if (selectedSectionId === id) setSelectedSectionId(null);
   286→  }
   287→
   288→  function moveSection(id: string, dir: -1 | 1) {
   289→    if (!config) return;
   290→    const idx = config.sections.findIndex((s) => s.id === id);
   291→    const target = idx + dir;
   292→    if (idx === -1 || target < 0 || target >= config.sections.length) return;
   293→    const sections = [...config.sections];
   294→    [sections[idx], sections[target]] = [sections[target], sections[idx]];
   295→    mutateConfig((c) => ({ ...c, sections }));
   296→  }
   297→
   298→  function updateSection(id: string, patch: Partial<SiteSection>) {
   299→    mutateConfig((c) => ({
   300→      ...c,
   301→      sections: c.sections.map((s) => (s.id === id ? ({ ...s, ...patch } as SiteSection) : s)),
   302→    }));
   303→  }
   304→
   305→  function updateTheme(patch: Partial<SiteTheme>) {
   306→    mutateConfig((c) => ({ ...c, theme: { ...c.theme, ...patch } }));
   307→  }
   308→
   309→  function applyTemplate(templateId: SiteConfig["template"]) {
   310→    const preset = templatePresets().find((t) => t.id === templateId);
   311→    if (!preset || !config) return;
   312→    mutateConfig((c) => ({
   313→      ...c,
   314→      template: preset.id,
   315→      theme: preset.theme,
   316→      sections: preset.sections.map((s) => ({ ...s })),
   317→    }));
   318→    toast.success(`${preset.label} template applied — remember to save`);
   319→    setSelectedSectionId(null);
   320→  }

   322→  // ------------------------------------------------------------ persistence

   324→  async function saveDraft({ silent = false } = {}): Promise<SiteConfig | null> {
   325→    if (!config) return null;
   326→    setSaving(true);
   327→    try {
   328→      await api("/api/me/site", { method: "PUT", json: { config } });
   329→      setDirty(false);
   330→      if (!silent) toast.success("Draft saved — only you can see it");
   331→      return config;
   332→    } catch (e) {
   333→      toast.error(e instanceof Error ? e.message : "Could not save the draft");
   334→      return null;
   335→    } finally {
   336→      setSaving(false);
   337→    }
   338→  }
   339→
   340→  async function publish() {
   341→    if (!config) return;
   342→    setPublishing(true);
   343→    try {
   344→      // Publishing snapshots the SAVED draft — flush unsaved edits first.
   345→      if (dirty) {
   346→        const saved = await saveDraft({ silent: true });
   347→        if (!saved) return;
   348→      }
   349→      const res = await api<{ status: string; publishedAt: string | null; publicUrl: string }>("/api/me/site/publish", { method: "POST" });
   350→      setDesign((d) =>
   351→        d
   352→          ? { ...d, status: "PUBLISHED", publishedAt: res.publishedAt, publicUrl: res.publicUrl, published: config, draft: config }
   353→          : d
   354→      );
   355→      toast.success("Website published — visitors now see it on your public profile");
   356→    } catch (e) {
   357→      toast.error(e instanceof Error ? e.message : "Could not publish");
   358→    } finally {
   359→      setPublishing(false);
   360→    }
   361→  }
   362→
   363→  async function unpublish() {
   364→    setPublishing(true);
   365→    try {
   366→      const res = await api<{ status: string }>("/api/me/site/unpublish", { method: "POST" });
   367→      setDesign((d) => (d ? { ...d, status: "DRAFT", publishedAt: null } : d));
   368→      toast.success("Website unpublished — visitors see the standard profile again");
   369→      void res;
   370→    } catch (e) {
   371→      toast.error(e instanceof Error ? e.message : "Could not unpublish");
   372→    } finally {
   373→      setPublishing(false);
   374→    }
   375→  }

   379→  function handleProposal(p: CodeGiftzProposal | null) {
   380→    if (!p) {

   388→  function acceptProposal() {

   391→    setConfig(proposal.config);
   392→    setDirty(true);
   393→    setProposal(null);
   394→    toast.success("Proposal applied as a draft — review and save");
   395→  }

   398→    setProposal(null);
   399→  }

   539→      {/* ---------------- work area */}
