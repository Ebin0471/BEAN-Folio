   176→/** Scroll-triggered reveal. Adds `bs-in` when the element enters the viewport. */
   177→function useReveal(motion: SiteMotion) {
   178→  const ref = useRef<HTMLDivElement | null>(null);
   179→  useEffect(() => {
   180→    const el = ref.current;
   181→    if (!el) return;
   182→    const reduced = typeof window !== "undefined" && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   183→    if (reduced || !motion.enabled || !motion.scrollReveal || motion.preset === "none") {
   184→      el.classList.add("bs-in");
   185→      return;
   186→    }
   187→    const io = new IntersectionObserver(
   188→      (entries) => {
   189→        for (const e of entries) {
   190→          if (e.isIntersecting) {
   191→            el.classList.add("bs-in");
   192→            io.disconnect();
   193→          }
   194→        }
   195→      },
   196→      { threshold: 0.12 }
   197→    );
   198→    io.observe(el);
   199→    return () => io.disconnect();
   200→  }, [motion]);
   201→  return ref;
   202→}
   203→
   204→function revealClass(motion: SiteMotion): string {
   205→  if (!motion.enabled || motion.preset === "none") return "";
   206→  switch (motion.preset) {
   207→    case "blurIn": return "bs-reveal bs-r-blurIn";
   208→    case "maskReveal": return "bs-reveal bs-r-maskReveal";
   209→    case "zoom": return "bs-reveal bs-r-zoom";
   210→    case "cinematic": return "bs-reveal bs-r-cinematic";
   211→    case "stagger": return "bs-reveal bs-r-fadeUp";
   212→    default: return "bs-reveal bs-r-fadeUp";
   213→  }
   214→}
   215→
   216→function Reveal({ motion, children, className, stagger = false }: { motion: SiteMotion; children: React.ReactNode; className?: string; stagger?: boolean }) {
   217→  const ref = useReveal(motion);
   218→  const style = {
   219→    ["--bs-dur" as string]: `${motion.duration}ms`,
   220→    ["--bs-ease" as string]: EASING[motion.easing],
   221→    ["--bs-int" as string]: String(motion.intensity),
   222→    ["--bs-stag" as string]: `${motion.stagger}ms`,
   223→  };
   224→  return (
   225→    <div ref={ref} style={style} className={cn(stagger && "bs-stagger", revealClass(motion), className)}>
   226→      {children}
   227→    </div>
   228→  );
   229→}

   231→/** Restrained hero-cover parallax (±~24px), desktop widths only. */
   232→function useParallax(enabled: boolean, intensity: number) {
   233→  const ref = useRef<HTMLImageElement | null>(null);
   234→  useEffect(() => {
   235→    const img = ref.current;
   236→    if (!img || !enabled) return;
   237→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   238→    if (reduced || window.innerWidth < 768) return;
   239→    let raf = 0;
   240→    const amount = Math.min(28, Math.max(6, (intensity / 100) * 28));
   241→    const update = () => {
   242→      raf = 0;
   243→      const rect = img.parentElement?.getBoundingClientRect();
   244→      if (!rect) return;
   245→      const progress = Math.min(1, Math.max(0, 1 - rect.top / Math.max(1, window.innerHeight)));
   246→      img.style.transform = `translateY(${(progress - 0.5) * amount * 2}px) scale(1.12)`;
   247→    };
   248→    const onScroll = () => {
   249→      if (!raf) raf = requestAnimationFrame(update);
   250→    };
   251→    update();
   252→    window.addEventListener("scroll", onScroll, { passive: true });
   253→    return () => {
   254→      window.removeEventListener("scroll", onScroll);
   255→      if (raf) cancelAnimationFrame(raf);
   256→    };
   257→  }, [enabled, intensity]);
   258→  return ref;
   259→}

   367→function SiteHero({ section, config, profile, theme, motion }: { section: Extract<SiteSection, { type: "hero" }>; config: SiteConfig; profile: SiteRendererProfile; theme: ResolvedTheme; motion: SiteMotion }) {
   368→  const scale = TYPE_SCALE[theme.typeScale];
   369→  const parallaxRef = useParallax(motion.enabled && motion.preset === "parallax", motion.intensity);
   370→  const variant: HeroVariant = section.variant ?? "classic";
   371→  const subtitle = fill(section.subtitle, profile, { tagline: config.tagline });
   372→  const showTagline = config.tagline && !section.subtitle.includes("{tagline}");
   373→
   374→  const headline = (
   375→    <h1
   376→      className={cn("max-w-3xl text-balance leading-[1.05]", scale.hero, variant === "centered" && "mx-auto text-center", variant === "split" && "max-w-none")}
   377→      style={{ fontWeight: HEADING_WEIGHT[theme.headingWeight], letterSpacing: TRACKING[theme.letterSpacing], fontFamily: FONT_STACKS[theme.headingFont] }}
   378→    >
   379→      {fill(section.title, profile, { tagline: config.tagline })}
   380→    </h1>
   381→  );
   382→
   383→  const cta = section.ctaLabel ? (
   384→    <a href="#bs-contact" className={buttonClasses(theme)} style={buttonStyle(theme)}>
   385→      {section.ctaLabel}
   386→    </a>
   387→  ) : null;
   388→
   389→  if (variant === "cover" && profile.coverUrl) {
   390→    return (
   391→      <section className="relative -mx-5 mb-4 overflow-hidden sm:-mx-8" style={{ minHeight: "min(72vh, 560px)" }}>
   392→        <img
   393→          ref={parallaxRef}
   394→          src={profile.coverUrl}
   395→          alt={`${profile.name} cover`}
   396→          className={cn("absolute inset-0 h-full w-full", motion.enabled && motion.preset === "parallax" && "bs-parallax")}
   397→          style={{ objectFit: "cover", objectPosition: `50% ${profile.coverPositionY}%`, transform: "scale(1.12)" }}
   398→        />
   399→        <div className="absolute inset-0" style={{ background: `linear-gradient(to top, ${theme.background} 8%, color-mix(in srgb, ${theme.background} 55%, transparent) 55%, transparent)` }} />
   400→        <div className="relative mx-auto flex w-full max-w-6xl flex-col justify-end px-5 pb-14 pt-40 sm:px-8">
   401→          {headline}
   402→          {subtitle && <p className="mt-4 max-w-2xl text-base opacity-85 sm:text-lg">{subtitle}</p>}
   403→          {showTagline ? <p className="mt-3 text-sm uppercase tracking-[0.2em] opacity-60">{config.tagline}</p> : null}
   404→          {cta && <div className="mt-6">{cta}</div>}
   405→        </div>
   406→      </section>
   407→    );
   408→  }
   409→
   410→  if (variant === "split") {
   411→    return (
   412→      <section className="grid items-end gap-8 pt-14 sm:grid-cols-[1.2fr_1fr] sm:pt-20">
   413→        {headline}
   414→        <div className="space-y-4 sm:pb-2">
   415→          {subtitle && <p className="text-base opacity-80 sm:text-lg">{subtitle}</p>}
   416→          {showTagline ? <p className="text-sm uppercase tracking-[0.2em] opacity-60">{config.tagline}</p> : null}
   417→          {cta}
   418→        </div>
   419→      </section>
   420→    );
   421→  }
   422→
   423→  return (
   424→    <section className={cn("pt-14 sm:pt-20", variant === "centered" && "text-center")}>
   425→      {profile.coverUrl && variant !== "cover" && (
   426→        <div className="mb-8 h-40 w-full overflow-hidden sm:h-56" style={{ borderRadius: theme.radius === "none" ? "0px" : "18px" }}>
   427→          <img
   428→            src={profile.coverUrl}
   429→            alt={`${profile.name} cover`}
   430→            className="h-full w-full"
   431→            style={{ objectFit: "cover", objectPosition: `50% ${profile.coverPositionY}%` }}
   432→          />
   433→        </div>
   434→      )}
   435→      {headline}
   436→      {subtitle && (
   437→        <p className={cn("mt-4 max-w-2xl text-base opacity-75 sm:text-lg", variant === "centered" && "mx-auto")}>
   438→          {subtitle}
   439→        </p>
   440→      )}
   441→      {showTagline ? (
   442→        <p className={cn("mt-3 text-sm uppercase tracking-[0.2em] opacity-60", variant === "centered" && "mx-auto")}>{config.tagline}</p>
   443→      ) : null}
   444→      {cta && <div className="mt-6">{cta}</div>}
   445→    </section>
   446→  );
   447→}

   451→export function SiteRenderer({ config, profile, projects, collections, onContact, pageId, onPageChange }: SiteRendererProps) {
   452→  const theme = useMemo(() => deriveTheme(config.theme), [config.theme]);
   453→  const motion = config.motion;
   454→  const pages = config.pages.length > 0 ? config.pages : [{ id: "home", name: "Home", visible: true, sections: [] } as SitePage];
   455→  const [internalPage, setInternalPage] = useState<string>(pages[0]?.id ?? "home");
   456→  const activePageId = pageId ?? internalPage;
   457→  const page = pages.find((p) => p.id === activePageId) ?? homePage(config) ?? pages[0];
   458→  const sections = (page?.sections ?? []).filter((s) => s && typeof s === "object" && s.visible !== false);
   459→  const navPages = pages.filter((p) => p.visible !== false);
   460→
   461→  const hoverZoom = motion.enabled && motion.hoverEffects;
   462→  const contentMax = CONTENT_WIDTH[theme.contentWidth];
