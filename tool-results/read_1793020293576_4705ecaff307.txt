    26→/** Mix two hex colors. t=0 → a, t=1 → b. Non-hex inputs return b. */
    27→export function mixHex(a: string, b: string, t: number): string {
    28→  const parse = (c: string): [number, number, number] | null => {
    29→    const m = /^#([0-9a-fA-F]{6})$/.exec(c.trim());
    30→    if (!m) return null;
    31→    const n = parseInt(m[1], 16);
    32→    return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
    33→  };
    34→  const pa = parse(a);
    35→  const pb = parse(b);
    36→  if (!pa || !pb) return /^#[0-9a-fA-F]{3,8}$/.test(b) ? b : "#888888";
    37→  const ch = (i: number) => Math.round(pa[i] + (pb[i] - pa[i]) * t);
    38→  const hex = (n: number) => n.toString(16).padStart(2, "0");
    39→  return `#${hex(ch(0))}${hex(ch(1))}${hex(ch(2))}`;
    40→}

    45→/** A theme after derivation: every token is a concrete string. */
    46→export interface ResolvedTheme extends Omit<SiteTheme, "surface" | "border" | "muted" | "headingFont" | "bodyFont"> {
    47→  surface: string;
    48→  border: string;
    49→  muted: string;
    50→  headingFont: SiteFontChoice;
    51→  bodyFont: SiteFontChoice;
    52→}
    53→
    54→/** Derived defaults for tokens a legacy/v1 config does not carry. */
    55→export function deriveTheme(theme: ThemeInput): ResolvedTheme {
    56→  const base = themeSchema.parse(theme);
    57→  return {
    58→    ...base,
    59→    surface: base.surface ?? mixHex(base.background, base.foreground, 0.06),
    60→    border: base.border ?? mixHex(base.background, base.foreground, 0.18),
    61→    muted: base.muted ?? mixHex(base.foreground, base.background, 0.42),
    62→    headingFont: base.headingFont ?? base.font,
    63→    bodyFont: base.bodyFont ?? base.font,
    64→  };
    65→}

   325→/** Defensive read helper — never let a missing/legacy `sections` array crash a render. */
   326→export function getVisibleSections(config: SiteConfig | null | undefined, pageId?: string): SiteSection[] {
   327→  if (!config || typeof config !== "object") return [];
   328→  const page = pageId ? config.pages.find((p) => p.id === pageId) : config.pages[0];
   329→  const sections = page?.sections ?? config.pages[0]?.sections ?? [];
   330→  return sections.filter((s) => s && typeof s === "object" && s.visible !== false);
   331→}
   332→
   333→/** The first visible page (used when no explicit page is selected). */
   334→export function homePage(config: SiteConfig): SitePage {
   335→  return config.pages.find((p) => p.visible) ?? config.pages[0];
   336→}

   340→/** Fresh section ids so duplicated presets never share React keys. */
   341→let uid = 0;
   342→function sid(type: string): string {
   343→  uid += 1;
   344→  return `${type}-${Date.now().toString(36)}-${uid}`;
   345→}

   347→export function makeSection(type: SiteSectionType): SiteSection {
   348→  const base = { id: sid(type), visible: true };
   349→  switch (type) {
   350→    case "hero":
   351→      return { ...base, type: "hero", title: "Selected work", subtitle: "", variant: "classic", ctaLabel: "" };
   352→    case "about":
   353→      return { ...base, type: "about", title: "About", body: "" };
   354→    case "projects":
   355→      return { ...base, type: "projects", title: "Projects", limit: 6, layout: "grid", category: "" };
   356→    case "skills":
   357→      return { ...base, type: "skills", title: "Skills & tools" };
   358→    case "stats":
   359→      return { ...base, type: "stats", title: "At a glance" };
   360→    case "collections":
   361→      return { ...base, type: "collections", title: "Collections", limit: 3 };
   362→    case "contact":
   363→      return { ...base, type: "contact", title: "Contact", body: "Available for commissions and collaborations." };
   364→    case "services":
   365→      return { ...base, type: "services", title: "Services", layout: "cards", items: [] };
   366→    case "experience":

   676→  return {
   677→    version: 1,
   678→    template: preset.id,
   679→    name: "My website",
   680→    tagline: "",
   681→    coverMediaId: null,
   682→    theme: deriveTheme(preset.theme),
   683→    motion: { ...defaultMotion(), ...(preset.motion ?? {}) },
   684→    pages: preset.pages.map((p) => ({ ...p, sections: p.sections.map((s) => ({ ...s })) })),
   685→  };
   686→}

   688→/**
   689→ * Normalize any parsed-but-possibly-incomplete config into a renderable one.
   690→ * Accepts an object OR a raw JSON string (as stored in SiteDesign.draftConfig).
   691→ * Migrates legacy v1 single-page configs into the `pages` model, applies
   692→ * derived theme tokens, and defaults the motion system.
   693→ * Returns null for anything that does not satisfy the schema — callers must
   694→ * treat null as "invalid/corrupt", never render it.
   695→ */
   696→export function normalizeConfig(input: unknown): SiteConfig | null {
   697→  let candidate = input;
   698→  if (typeof candidate === "string") {
   699→    try {
   700→      candidate = JSON.parse(candidate);
   701→    } catch {
   702→      return null;
   703→    }
   704→  }
