    50→interface AiFailure {
    51→  title: string;
    52→  hint?: string;
    53→  /** "retry" → same request can succeed; "settings" → user action required elsewhere. */
    54→  action?: "retry" | "settings";
    55→}
    56→
    57→/** Map any thrown value to an honest, user-safe failure description. Never fakes success. */
    58→function aiFailure(e: unknown): AiFailure {
    59→  if (e instanceof ApiError) {
    60→    if (e.status === 403) {
    61→      return {
    62→        title: e.message,
    63→        hint: "You can enable BEAN Intelligence on this page or in Settings → Privacy.",
    64→        action: "settings",
    65→      };
    66→    }
    67→    if (e.status === 429) {
    68→      return { title: e.message, hint: "Daily usage resets at midnight UTC.", action: "retry" };
    69→    }
    70→    if (e.status === 502) {
    71→      return { title: "The AI service is temporarily unavailable.", hint: e.message, action: "retry" };
    72→    }
    73→    if (e.status === 0) {
    74→      return { title: "Network error — check your connection and try again.", action: "retry" };
    75→    }
    76→    return { title: e.message, action: "retry" };
    77→  }
    78→  return { title: "Something went wrong. Please try again.", action: "retry" };
    79→}

   231→function ChatTab({ consented, onQuotaUsed }: { consented: boolean; onQuotaUsed: () => void }) {
   232→  const [conversations, setConversations] = useState<AiConversationSummary[]>([]);
   233→  const [convLoading, setConvLoading] = useState(false);
   234→  const [activeId, setActiveId] = useState<string | null>(null);
   235→  const [thread, setThread] = useState<AiMessageDTO[]>([]);
   236→  const [threadLoading, setThreadLoading] = useState(false);
   237→  const [sending, setSending] = useState(false);
   238→  const [input, setInput] = useState("");
   239→  const [failure, setFailure] = useState<AiFailure | null>(null);
   240→  const threadRef = useRef<HTMLDivElement>(null);
   241→
   242→  const refreshConversations = useCallback(async (silent = false) => {
   243→    if (!silent) setConvLoading(true);
   244→    try {
   245→      const res = await api<{ conversations: AiConversationSummary[] }>("/api/ai/conversations");
   246→      setConversations(res.conversations);
   247→    } catch (e) {
   248→      // A 403 here means consent was revoked — keep the list empty, never fake data.
   249→      if (!(e instanceof ApiError && e.status === 403)) {
   250→        if (!silent) toast.error(e instanceof ApiError ? e.message : "Couldn't load conversations.");
   251→      }
   252→    } finally {
   253→      if (!silent) setConvLoading(false);
   254→    }
   255→  }, []);
   256→
   257→  useEffect(() => {
   258→    if (consented) void refreshConversations();
   259→  }, [consented, refreshConversations]);
   260→
   261→  useEffect(() => {
   262→    const el = threadRef.current;
   263→    if (el) el.scrollTop = el.scrollHeight;
   264→  }, [thread, sending]);
   265→
   266→  const openConversation = async (id: string) => {
   267→    if (sending) return;
   268→    setActiveId(id);
   269→    setThreadLoading(true);
   270→    setFailure(null);
   271→    try {
   272→      const res = await api<{ conversation: { id: string; title: string; messages: AiMessageDTO[] } }>(
   273→        `/api/ai/conversations/${id}`
   274→      );
   275→      setThread(res.conversation.messages);
   276→    } catch (e) {
   277→      setThread([]);
   278→      setFailure(aiFailure(e));
   279→    } finally {
   280→      setThreadLoading(false);
   281→    }
   282→  };
   283→
   284→  const newChat = () => {
   285→    if (sending) return;
   286→    setActiveId(null);
   287→    setThread([]);
   288→    setFailure(null);
   289→    setInput("");
   290→  };
   291→
   292→  const send = async (raw: string) => {
   293→    const message = raw.trim();
   294→    if (!message || sending) return;
   295→    setFailure(null);
   296→    setSending(true);
   297→    const optimistic: AiMessageDTO = { role: "user", content: message, createdAt: new Date().toISOString() };
   298→    setThread((t) => [...t, optimistic]);
   299→    setInput("");
   300→    try {
   301→      const res = await api<{ conversationId: string; reply: string }>("/api/ai/chat", {
   302→        method: "POST",
   303→        json: { conversationId: activeId ?? undefined, message },
   304→      });
   305→      setThread((t) => [
   306→        ...t,
   307→        { role: "assistant", content: res.reply, createdAt: new Date().toISOString() },
   308→      ]);
   309→      setActiveId(res.conversationId);
   310→      onQuotaUsed();
   311→      void refreshConversations(true);
   312→    } catch (e) {
   313→      // The optimistic message was never persisted — remove it so the thread stays honest.
   314→      setThread((t) => t.filter((m) => m !== optimistic));
   315→      setInput(message);
   316→      setFailure(aiFailure(e));
   317→    } finally {
   318→      setSending(false);
   319→    }
   320→  };
   321→
   322→  const mobilePicker = (
   323→    <div className="mb-4 flex items-center gap-2 md:hidden">
   324→      <Select
   325→        value={activeId ?? "__new"}
   326→        onValueChange={(v) => (v === "__new" ? newChat() : void openConversation(v))}
   327→      >
   328→        <SelectTrigger className="h-10 flex-1" aria-label="Choose a conversation">
   329→          <SelectValue placeholder="Conversations" />
   330→        </SelectTrigger>
   331→        <SelectContent>
   332→          <SelectItem value="__new">＋ New chat</SelectItem>
   333→          {conversations.map((c) => (
   334→            <SelectItem key={c.id} value={c.id}>
   335→              {c.title || "Untitled conversation"}
   336→            </SelectItem>
   337→          ))}
   338→        </SelectContent>
   339→      </Select>
   340→      <Button variant="outline" size="icon" className="size-10 shrink-0" onClick={newChat} aria-label="Start a new chat">
   341→        <Plus className="size-4" aria-hidden />
   342→      </Button>
   343→    </div>
   344→  );
   345→
   346→  return (
   347→    <div className="grid gap-4 md:grid-cols-[260px_1fr] md:gap-6">
   348→      {mobilePicker}
   349→
   350→      {/* Sidebar (desktop) */}
   351→      <aside className="hidden md:block">
   352→        <div className="flex items-center justify-between pb-2">
   353→          <h3 className="text-sm font-medium text-foreground">Conversations</h3>
   354→          <Button variant="outline" size="sm" onClick={newChat}>
   355→            <Plus className="size-3.5" aria-hidden /> New
   356→          </Button>
   357→        </div>
   358→        <div className="max-h-96 space-y-1.5 overflow-y-auto pr-1" aria-busy={convLoading}>
   359→          {convLoading ? (
   360→            <div className="space-y-2 py-1">
   361→              <Skeleton className="h-12 w-full" />
   362→              <Skeleton className="h-12 w-full" />
   363→              <Skeleton className="h-12 w-full" />
   364→            </div>
   365→          ) : conversations.length === 0 ? (
   366→            <p className="px-1 py-3 text-xs leading-relaxed text-muted-foreground">
   367→              No conversations yet. Your chats with the assistant will appear here.
   368→            </p>
   369→          ) : (
   370→            conversations.map((c) => (
   371→              <button
   372→                key={c.id}
   373→                type="button"
   374→                onClick={() => void openConversation(c.id)}
   375→                aria-current={activeId === c.id ? "true" : undefined}
   376→                className={cn(
   377→                  "block w-full rounded-lg border px-3 py-2.5 text-left transition-colors",
   378→                  activeId === c.id
   379→                    ? "border-primary/40 bg-gold-soft"
   380→                    : "border-border/50 bg-card hover:border-primary/25 hover:bg-accent"
   381→                )}
   382→              >
   383→                <p className="truncate text-sm font-medium text-foreground">{c.title || "Untitled conversation"}</p>
   384→                <p className="mt-0.5 text-[11px] text-muted-foreground">{formatDate(c.updatedAt)}</p>
   385→              </button>
   386→            ))
   387→          )}
   388→        </div>
   389→      </aside>
   390→
   391→      {/* Thread */}

   399→                  Open Settings

   633→function SuggestToolCard({ onQuotaUsed }: { onQuotaUsed: () => void }) {
   634→  const [title, setTitle] = useState("");
   635→  const [description, setDescription] = useState("");
   636→  const [loading, setLoading] = useState(false);
   637→  const [result, setResult] = useState<SuggestResult | null>(null);
   638→  const [failure, setFailure] = useState<AiFailure | null>(null);
   639→
   640→  const run = async () => {
   641→    if (!title.trim() || loading) return;
   642→    setLoading(true);
   643→    setFailure(null);
   644→    try {
   645→      const res = await api<SuggestResult>("/api/ai/suggest", {
   646→        method: "POST",
   647→        json: { title: title.trim(), description: description.trim() || undefined },
   648→      });
   649→      setResult(res);
   650→      onQuotaUsed();
   651→      toast.success("Suggestions ready — copy what you like");
   652→    } catch (e) {
   653→      setResult(null);
   654→      setFailure(aiFailure(e));
   655→    } finally {
   656→      setLoading(false);
   657→    }
   658→  };
   659→
   660→  return (
   661→    <Card>
   662→      <CardContent className="p-6">
   663→        <div className="flex items-center gap-3">
   664→          <div className="grid size-9 place-items-center rounded-lg bg-gold-soft text-primary ring-1 ring-primary/25">
   665→            <Lightbulb className="size-4" aria-hidden />

   667→          <div>
   668→            <h3 className="font-display text-lg font-semibold text-foreground">Suggest metadata</h3>
   669→            <p className="text-xs text-muted-foreground">Propose a summary, skills and a profession category.</p>
   670→          </div>
   671→        </div>
   672→
   673→        <div className="mt-5 space-y-4">
   674→          <div className="space-y-1.5">

   678→              value={title}
   679→              onChange={(e) => setTitle(e.target.value)}
   680→              maxLength={140}
   681→              placeholder="e.g. Tidelines — a harbour restoration photo essay"
   682→            />
   683→          </div>
   684→          <div className="space-y-1.5">
   685→            <Label htmlFor="suggest-description">Description</Label>
   686→            <Textarea
   687→              id="suggest-description"
   688→              value={description}
   689→              onChange={(e) => setDescription(e.target.value)}
   690→              rows={4}
   691→              maxLength={6000}
   692→              placeholder="Describe the project briefly — even a rough draft helps."
   693→              className="resize-y"
   694→            />
   695→          </div>
   696→
   697→          {failure && <InlineError failure={failure} onRetry={failure.action === "retry" ? () => void run() : undefined} />}
   698→
   699→          <div className="flex flex-wrap items-center gap-3">
   700→            <Button onClick={() => void run()} disabled={loading || !title.trim()}>
   701→              {loading ? <Spinner /> : <Sparkles className="size-4" aria-hidden />}
   702→              {loading ? "Suggesting…" : "Suggest"}
   703→            </Button>
   704→            <p className="text-xs text-muted-foreground">Nothing is saved to your portfolio automatically.</p>
   705→          </div>
   706→
   707→          {result && (
   708→            <div className="space-y-4 rounded-lg border border-primary/25 bg-background/40 p-4">
   709→              {result.summary ? (

   711→                  <div className="mb-1.5 flex items-center justify-between gap-2">

   716→                    {result.summary}
   717→                  </p>
   718→                </div>
   719→              ) : null}
   720→

   722→                <div>

   726→                  </div>

   729→                      <Badge key={s} variant="outline" className="rounded-full text-xs font-normal">
   730→                        {s}
   731→                      </Badge>
   732→                    ))}
   733→                  </div>
   734→                </div>
   735→              ) : null}

   738→                <div>
   739→                  <div className="mb-1.5 flex items-center justify-between gap-2">
   740→                    <p className="text-xs font-semibold uppercase tracking-wide text-primary">Suggested category</p>
