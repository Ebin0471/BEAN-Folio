   176→   176→/** Scroll-triggered reveal. Adds `bs-in` when the element enters the viewport. */
   177→   177→function useReveal(motion: SiteMotion) {
   178→   178→  const ref = useRef<HTMLDivElement | null>(null);
   179→   179→  useEffect(() => {
   180→   180→    const el = ref.current;
   181→   181→    if (!el) return;
   182→   182→    const reduced = typeof window !== "undefined" && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   183→   183→    if (reduced || !motion.enabled || !motion.scrollReveal || motion.preset === "none") {
   184→   184→      el.classList.add("bs-in");
   185→   185→      return;
   186→   186→    }
   187→   187→    const io = new IntersectionObserver(
   188→   188→      (entries) => {
   189→   189→        for (const e of entries) {
   190→   190→          if (e.isIntersecting) {
   191→   191→            el.classList.add("bs-in");
   192→   192→            io.disconnect();
   193→   193→          }
   194→   194→        }
   195→   195→      },
   196→   196→      { threshold: 0.12 }
   197→   197→    );
   198→   198→    io.observe(el);
   199→   199→    return () => io.disconnect();
   200→   200→  }, [motion]);
   201→   201→  return ref;
   202→   202→}

   204→   204→function revealClass(motion: SiteMotion): string {
   205→   205→  if (!motion.enabled || motion.preset === "none") return "";
   206→   206→  switch (motion.preset) {
   207→   207→    case "blurIn": return "bs-reveal bs-r-blurIn";
   208→   208→    case "maskReveal": return "bs-reveal bs-r-maskReveal";
   209→   209→    case "zoom": return "bs-reveal bs-r-zoom";
   210→   210→    case "cinematic": return "bs-reveal bs-r-cinematic";
   211→   211→    case "stagger": return "bs-reveal bs-r-fadeUp";
   212→   212→    default: return "bs-reveal bs-r-fadeUp";
   213→   213→  }
   214→   214→}
   215→   215→
   216→   216→function Reveal({ motion, children, className, stagger = false }: { motion: SiteMotion; children: React.ReactNode; className?: string; stagger?: boolean }) {
   217→   217→  const ref = useReveal(motion);
   218→   218→  const style = {
   219→   219→    ["--bs-dur" as string]: `${motion.duration}ms`,
   220→   220→    ["--bs-ease" as string]: EASING[motion.easing],
   221→   221→    ["--bs-int" as string]: String(motion.intensity),
   222→   222→    ["--bs-stag" as string]: `${motion.stagger}ms`,
   223→   223→  };
   224→   224→  return (
   225→   225→    <div ref={ref} style={style} className={cn(stagger && "bs-stagger", revealClass(motion), className)}>
   226→   226→      {children}
   227→   227→    </div>
   228→   228→  );
   229→   229→}

   231→   231→/** Restrained hero-cover parallax (±~24px), desktop widths only. */
   232→   232→function useParallax(enabled: boolean, intensity: number) {
   233→   233→  const ref = useRef<HTMLImageElement | null>(null);
   234→   234→  useEffect(() => {
   235→   235→    const img = ref.current;
   236→   236→    if (!img || !enabled) return;
   237→   237→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   238→   238→    if (reduced || window.innerWidth < 768) return;
   239→   239→    let raf = 0;
   240→   240→    const amount = Math.min(28, Math.max(6, (intensity / 100) * 28));
   241→   241→    const update = () => {
   242→   242→      raf = 0;
   243→   243→      const rect = img.parentElement?.getBoundingClientRect();
   244→   244→      if (!rect) return;
   245→   245→      const progress = Math.min(1, Math.max(0, 1 - rect.top / Math.max(1, window.innerHeight)));
   246→   246→      img.style.transform = `translateY(${(progress - 0.5) * amount * 2}px) scale(1.12)`;
   247→   247→    };
   248→   248→    const onScroll = () => {
   249→   249→      if (!raf) raf = requestAnimationFrame(update);
   250→   250→    };
   251→   251→    update();
   252→   252→    window.addEventListener("scroll", onScroll, { passive: true });
   253→   253→    return () => {
   254→   254→      window.removeEventListener("scroll", onScroll);
   255→   255→      if (raf) cancelAnimationFrame(raf);
   256→   256→    };
   257→   257→  }, [enabled, intensity]);
   258→   258→  return ref;
   259→   259→}

   367→   367→function SiteHero({ section, config, profile, theme, motion }: { section: Extract<SiteSection, { type: "hero" }>; config: SiteConfig; profile: SiteRendererProfile; theme: ResolvedTheme; motion: SiteMotion }) {
   368→   368→  const scale = TYPE_SCALE[theme.typeScale];
   369→   369→  const parallaxRef = useParallax(motion.enabled && motion.preset === "parallax", motion.intensity);
   370→   370→  const variant: HeroVariant = section.variant ?? "classic";
   371→   371→  const subtitle = fill(section.subtitle, profile, { tagline: config.tagline });
   372→   372→  const showTagline = config.tagline && !section.subtitle.includes("{tagline}");
   373→   373→
   374→   374→  const headline = (
   375→   375→    <h1
   376→   376→      className={cn("max-w-3xl text-balance leading-[1.05]", scale.hero, variant === "centered" && "mx-auto text-center", variant === "split" && "max-w-none")}
   377→   377→      style={{ fontWeight: HEADING_WEIGHT[theme.headingWeight], letterSpacing: TRACKING[theme.letterSpacing], fontFamily: FONT_STACKS[theme.headingFont] }}
   378→   378→    >
   379→   379→      {fill(section.title, profile, { tagline: config.tagline })}
   380→   380→    </h1>
   381→   381→  );
   382→   382→
   383→   383→  const cta = section.ctaLabel ? (
   384→   384→    <a href="#bs-contact" className={buttonClasses(theme)} style={buttonStyle(theme)}>
   385→   385→      {section.ctaLabel}
   386→   386→    </a>
   387→   387→  ) : null;
   388→   388→
   389→   389→  if (variant === "cover" && profile.coverUrl) {
   390→   390→    return (
   391→   391→      <section className="relative -mx-5 mb-4 overflow-hidden sm:-mx-8" style={{ minHeight: "min(72vh, 560px)" }}>
   392→   392→        <img
   393→   393→          ref={parallaxRef}
   394→   394→          src={profile.coverUrl}
   395→   395→          alt={`${profile.name} cover`}
   396→   396→          className={cn("absolute inset-0 h-full w-full", motion.enabled && motion.preset === "parallax" && "bs-parallax")}

   444→   444→      {cta && <div className="mt-6">{cta}</div>}
   445→   445→    </section>
   446→   446→  );
