     1→/**
     2→ * BEAN Folio — demo fixture seeder.
     3→ *
     4→ * Creates CLEARLY-MARKED demo accounts (User.isDemo = true) and fixture
     5→ * content so the product can be explored with realistic data. This is demo
     6→ * data, not production content: it is only created when this script is run
     7→ * explicitly (`bun run scripts/seed.ts`), and `bun run scripts/seed.ts --clear`
     8→ * removes every isDemo account and its content (real accounts are never touched).
     9→ *
    10→ * Passwords are hashed with the same scrypt scheme as src/lib/auth.ts.
    11→ */

    21→async function hashPassword(password: string): Promise<string> {
    22→  const salt = randomBytes(16).toString("hex");
    23→  const derived = await scrypt(password, salt, 64);
    24→  return `${salt}:${derived.toString("hex")}`;
    25→}

    34→let assetCounter = 0;
    35→async function importImage(ownerId: string, file: string, filename: string) {
    36→  const src = path.join(ASSETS, file);
    37→  if (!existsSync(src)) throw new Error(`Missing seed asset: ${file}`);
    38→  mkdirSync(UPLOADS, { recursive: true });
    39→  assetCounter += 1;
    40→  const name = `seed-${assetCounter}-${file}`; // unique per asset row — rows never share files
    41→  copyFileSync(src, path.join(UPLOADS, name));
    42→  const size = statSync(src).size;
    43→  return db.mediaAsset.create({
    44→    data: { ownerId, kind: "IMAGE", filename, mime: "image/png", size, status: "READY", storagePath: path.join("uploads", name) },
    45→  });
    46→}
