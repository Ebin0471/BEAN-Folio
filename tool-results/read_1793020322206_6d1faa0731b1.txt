   176→/** Scroll-triggered reveal. Adds `bs-in` when the element enters the viewport. */
   177→function useReveal(motion: SiteMotion) {
   178→  const ref = useRef<HTMLDivElement | null>(null);
   179→  useEffect(() => {
   180→    const el = ref.current;
   181→    if (!el) return;
   182→    const reduced = typeof window !== "undefined" && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   183→    if (reduced || !motion.enabled || !motion.scrollReveal || motion.preset === "none") {
   184→      el.classList.add("bs-in");
   185→      return;
   186→    }
   187→    const io = new IntersectionObserver(
   188→      (entries) => {
   189→        for (const e of entries) {
   190→          if (e.isIntersecting) {
   191→            el.classList.add("bs-in");
   192→            io.disconnect();
   193→          }
   194→        }
   195→      },
   196→      { threshold: 0.12 }
   197→    );
   198→    io.observe(el);
   199→    return () => io.disconnect();
   200→  }, [motion]);
   201→  return ref;
   202→}
   203→
   204→function revealClass(motion: SiteMotion): string {
   205→  if (!motion.enabled || motion.preset === "none") return "";
   206→  switch (motion.preset) {
   207→    case "blurIn": return "bs-reveal bs-r-blurIn";
   208→    case "maskReveal": return "bs-reveal bs-r-maskReveal";
   209→    case "zoom": return "bs-reveal bs-r-zoom";
   210→    case "cinematic": return "bs-reveal bs-r-cinematic";
   211→    case "stagger": return "bs-reveal bs-r-fadeUp";
   212→    default: return "bs-reveal bs-r-fadeUp";
   213→  }
   214→}
   215→
   216→function Reveal({ motion, children, className, stagger = false }: { motion: SiteMotion; children: React.ReactNode; className?: string; stagger?: boolean }) {
   217→  const ref = useReveal(motion);
   218→  const style = {
   219→    ["--bs-dur" as string]: `${motion.duration}ms`,
   220→    ["--bs-ease" as string]: EASING[motion.easing],
   221→    ["--bs-int" as string]: String(motion.intensity),
   222→    ["--bs-stag" as string]: `${motion.stagger}ms`,
   223→  };
   224→  return (
   225→    <div ref={ref} style={style} className={cn(stagger && "bs-stagger", revealClass(motion), className)}>
   226→      {children}
   227→    </div>
   228→  );
   229→}

   231→/** Restrained hero-cover parallax (±~24px), desktop widths only. */
   232→function useParallax(enabled: boolean, intensity: number) {
   233→  const ref = useRef<HTMLImageElement | null>(null);
   234→  useEffect(() => {
   235→    const img = ref.current;
   236→    if (!img || !enabled) return;
   237→    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   238→    if (reduced || window.innerWidth < 768) return;
   239→    let raf = 0;
   240→    const amount = Math.min(28, Math.max(6, (intensity / 100) * 28));
   241→    const update = () => {
   242→      raf = 0;
   243→      const rect = img.parentElement?.getBoundingClientRect();
   244→      if (!rect) return;
   245→      const progress = Math.min(1, Math.max(0, 1 - rect.top / Math.max(1, window.innerHeight)));
   246→      img.style.transform = `translateY(${(progress - 0.5) * amount * 2}px) scale(1.12)`;
   247→    };
   248→    const onScroll = () => {
   249→      if (!raf) raf = requestAnimationFrame(update);
   250→    };
   251→    update();
   252→    window.addEventListener("scroll", onScroll, { passive: true });
   253→    return () => {
   254→      window.removeEventListener("scroll", onScroll);
   255→      if (raf) cancelAnimationFrame(raf);
   256→    };
   257→  }, [enabled, intensity]);
   258→  return ref;
   259→}

   325→/** True only for a usable image URL — an empty/blank src must never reach an <img>. */
   326→function hasImageSrc(src: string | null | undefined): src is string {
   327→  return typeof src === "string" && src.trim().length > 0;
   328→}

   330→/**
   331→ * Cover image with a theme-aware fallback. If the cover is missing or blank,
   332→ * a decorative monogram tile is rendered using the site's own theme tokens —
   333→ * no <img> element at all, so no empty-src request and no console error.
   334→ * The fallback is aria-hidden: the enclosing card/link already exposes the
   335→ * project title as accessible text.
   336→ */
   337→function SiteCoverImage({
   338→  src, alt, title, theme,
   339→}: {
   340→  src: string | null | undefined;
   341→  alt: string;
   342→  title: string;
   343→  theme: ResolvedTheme;
   344→}) {
   345→  if (!hasImageSrc(src)) {
   346→    const monogram = title.trim().charAt(0).toUpperCase() || "•";
   347→    return (
   348→      <div
   349→        className="flex h-full w-full items-center justify-center"
   350→        style={{
   351→          background: `linear-gradient(145deg, color-mix(in srgb, ${theme.accent} 14%, ${theme.background}) 0%, ${theme.surface} 58%, ${theme.background} 100%)`,
   352→        }}
   353→        aria-hidden="true"
   354→      >
   355→        <span
   356→          className="select-none leading-none"
   357→          style={{
   358→            color: theme.accent,
   359→            opacity: 0.55,
   360→            fontFamily: FONT_STACKS[theme.headingFont],
   361→            fontWeight: HEADING_WEIGHT[theme.headingWeight],
   362→            letterSpacing: TRACKING[theme.letterSpacing],
   363→            fontSize: "clamp(1.75rem, 5vw, 3.5rem)",
   364→          }}
   365→        >
   366→          {monogram}
   367→        </span>
   368→      </div>
   369→    );
   370→  }
   371→  return <img src={src} alt={alt} className="h-full w-full" style={{ objectFit: "cover" }} loading="lazy" />;
   372→}

   412→function SiteHero({ section, config, profile, theme, motion }: { section: Extract<SiteSection, { type: "hero" }>; config: SiteConfig; profile: SiteRendererProfile; theme: ResolvedTheme; motion: SiteMotion }) {
   413→  const scale = TYPE_SCALE[theme.typeScale];
   414→  const parallaxRef = useParallax(motion.enabled && motion.preset === "parallax", motion.intensity);
   415→  const variant: HeroVariant = section.variant ?? "classic";
   416→  const subtitle = fill(section.subtitle, profile, { tagline: config.tagline });
   417→  const showTagline = config.tagline && !section.subtitle.includes("{tagline}");
   418→  const coverSrc = hasImageSrc(profile.coverUrl) ? profile.coverUrl : null;
   419→
   420→  const headline = (
   421→    <h1
   422→      className={cn("max-w-3xl text-balance leading-[1.05]", scale.hero, variant === "centered" && "mx-auto text-center", variant === "split" && "max-w-none")}
   423→      style={{ fontWeight: HEADING_WEIGHT[theme.headingWeight], letterSpacing: TRACKING[theme.letterSpacing], fontFamily: FONT_STACKS[theme.headingFont] }}
   424→    >
   425→      {fill(section.title, profile, { tagline: config.tagline })}
   426→    </h1>
   427→  );
   428→
   429→  const cta = section.ctaLabel ? (
   430→    <a href="#bs-contact" className={buttonClasses(theme)} style={buttonStyle(theme)}>
   431→      {section.ctaLabel}
   432→    </a>
   433→  ) : null;
   434→
   435→  if (variant === "cover" && coverSrc) {
   436→    return (
   437→      <section className="relative -mx-5 mb-4 overflow-hidden sm:-mx-8" style={{ minHeight: "min(72vh, 560px)" }}>
   438→        <img
   439→          ref={parallaxRef}
   440→          src={coverSrc}
   441→          alt={`${profile.name} cover`}
   442→          className={cn("absolute inset-0 h-full w-full", motion.enabled && motion.preset === "parallax" && "bs-parallax")}
   443→          style={{ objectFit: "cover", objectPosition: `50% ${profile.coverPositionY}%`, transform: "scale(1.12)" }}
   444→        />
   445→        <div className="absolute inset-0" style={{ background: `linear-gradient(to top, ${theme.background} 8%, color-mix(in srgb, ${theme.background} 55%, transparent) 55%, transparent)` }} />
   446→        <div className="relative mx-auto flex w-full max-w-6xl flex-col justify-end px-5 pb-14 pt-40 sm:px-8">
   447→          {headline}
   448→          {subtitle && <p className="mt-4 max-w-2xl text-base opacity-85 sm:text-lg">{subtitle}</p>}
   449→          {showTagline ? <p className="mt-3 text-sm uppercase tracking-[0.2em] opacity-60">{config.tagline}</p> : null}
   450→          {cta && <div className="mt-6">{cta}</div>}
   451→        </div>
   452→      </section>
   453→    );
   454→  }
   455→
   456→  if (variant === "split") {
   457→    return (
   458→      <section className="grid items-end gap-8 pt-14 sm:grid-cols-[1.2fr_1fr] sm:pt-20">
   459→        {headline}
   460→        <div className="space-y-4 sm:pb-2">
   461→          {subtitle && <p className="text-base opacity-80 sm:text-lg">{subtitle}</p>}
   462→          {showTagline ? <p className="text-sm uppercase tracking-[0.2em] opacity-60">{config.tagline}</p> : null}
   463→          {cta}
   464→        </div>
   465→      </section>
   466→    );
   467→  }
   468→
   469→  return (
   470→    <section className={cn("pt-14 sm:pt-20", variant === "centered" && "text-center")}>
   471→      {coverSrc && variant !== "cover" && (

   474→            src={coverSrc}
   475→            alt={`${profile.name} cover`}
   476→            className="h-full w-full"
   477→            style={{ objectFit: "cover", objectPosition: `50% ${profile.coverPositionY}%` }}
   478→          />
   479→        </div>
   480→      )}
   481→      {headline}
   482→      {subtitle && (
   483→        <p className={cn("mt-4 max-w-2xl text-base opacity-75 sm:text-lg", variant === "centered" && "mx-auto")}>
   484→          {subtitle}
   485→        </p>
   486→      )}
   487→      {showTagline ? (
   488→        <p className={cn("mt-3 text-sm uppercase tracking-[0.2em] opacity-60", variant === "centered" && "mx-auto")}>{config.tagline}</p>
   489→      ) : null}
   490→      {cta && <div className="mt-6">{cta}</div>}
   491→    </section>
   492→  );
   493→}
